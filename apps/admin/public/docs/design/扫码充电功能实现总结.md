#### 后端接口实现：
```typescript
// apps/server/src/services/pileService.ts
import QRCode from 'qrcode';

export async function generatePileQRCode(pileId: number): Promise<Buffer> {
  // 生成二维码内容：PILE_{pileId}
  const qrCodeContent = `PILE_${pileId}`;
  
  // 生成二维码图片（PNG格式）
  const qrCodeBuffer = await QRCode.toBuffer(qrCodeContent, {
    type: 'png',
    width: 300,
    margin: 2,
    color: {
      dark: '#000000',
      light: '#FFFFFF',
    },
  });
  
  return qrCodeBuffer;
}
```

#### 路由实现：
```typescript
// apps/server/src/routes/pileRoutes.ts
router.get('/piles/:id/qrcode', authMiddleware, async (req, res) => {
  try {
    const pileId = parseInt(req.params.id);
    const qrCodeBuffer = await generatePileQRCode(pileId);
    
    res.setHeader('Content-Type', 'image/png');
    res.setHeader('Content-Disposition', `attachment; filename="pile_${pileId}_qrcode.png"`);
    res.send(qrCodeBuffer);
  } catch (err: any) {
    res.status(500).json({ code: 500, message: err.message });
  }
});
```

  window.URL.revokeObjectURL(url);

### 2. 充电中页面优化

#### WebSocket连接优化：
```typescript
// apps/user-app/src/store/modules/charging.ts
function connectWebSocket(orderId: string) {
  if (wsConnection.value) {
    disconnectWebSocket();
  }
  
  const token = uni.getStorageSync('user_token');
  let reconnectCount = 0;
  const maxReconnect = 5;
  
  function connect() {
    wsConnection.value = uni.connectSocket({
      url: `${config.wsUrl}/charging/${orderId}?token=${token}`,
      success: () => {
        console.log('WebSocket 连接成功');
        reconnectCount = 0;
      },
      fail: () => {
        // 连接失败，切换到轮询模式
        startPolling();
      },
    });
    
    // 监听消息
    uni.onSocketMessage((res) => {
      try {
        const data = JSON.parse(res.data as string);
        if (data.type === 'charging_status') {
          chargingStatus.value = {
            ...chargingStatus.value,
            ...data.payload,
          } as ChargingStatus;
        }
      } catch (e) {
        console.error('WebSocket 消息解析失败', e);
      }
    });
    
    // 监听关闭
    uni.onSocketClose(() => {
      console.log('WebSocket 连接关闭');
      wsConnection.value = null;
      
      // 尝试重连
      if (reconnectCount < maxReconnect) {
        reconnectCount++;
        setTimeout(connect, 3000 * reconnectCount);
      } else {
        // 重连失败，切换到轮询模式
        startPolling();
      }
    });
    
    // 监听错误
    uni.onSocketError(() => {
      console.error('WebSocket 连接错误');
      wsConnection.value = null;
      startPolling();
    });
  }
  
  function startPolling() {
    // 如果WebSocket失败，使用HTTP轮询
    const pollInterval = setInterval(async () => {
      try {
        const status = await chargingApi.getChargingStatus();
        if (status) {
          chargingStatus.value = status;
        } else {
          clearInterval(pollInterval);
          isCharging.value = false;
        }
      } catch (error) {
        console.error('轮询获取充电状态失败', error);
      }
    }, 3000);
    
    // 存储轮询ID，便于清理
    (wsConnection.value as any).pollInterval = pollInterval;
  }
  
  connect();
}
```
