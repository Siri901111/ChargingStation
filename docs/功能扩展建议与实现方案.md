##### 后端实现

**1. 日志中间件**
```typescript
// apps/server/src/middlewares/logMiddleware.ts
import { Request, Response, NextFunction } from 'express';
import SystemLog from '../models/SystemLog';

export function logMiddleware(req: Request, res: Response, next: NextFunction) {
  const startTime = Date.now();
  const userId = (req as any).userId;
  const userAccount = (req as any).userAccount;
  
  // 获取原始响应方法
  const originalJson = res.json;
  res.json = function(body: any) {
    const duration = Date.now() - startTime;
    
    // 异步记录日志（不阻塞响应）
    setImmediate(async () => {
      try {
        await SystemLog.create({
          user_id: userId,
          user_account: userAccount,
          action: extractAction(req.path, req.method),
          module: extractModule(req.path),
          target_id: req.params.id || null,
          request_method: req.method,
          request_path: req.path,
          request_params: JSON.stringify(req.body || req.query),
          response_code: res.statusCode,
          response_message: body.message || null,
          ip_address: req.ip || req.headers['x-forwarded-for'] as string,
          user_agent: req.headers['user-agent'],
          duration,
          status: res.statusCode >= 400 ? 2 : 1,
          error_message: body.code !== 200 ? body.message : null,
        });
      } catch (error) {
        console.error('记录日志失败:', error);
      }
    });
    
    return originalJson.call(this, body);
  };
  
  next();
}

function extractAction(path: string, method: string): string {
  if (path.includes('/login')) return 'login';
  if (method === 'POST') return 'create';
  if (method === 'PUT' || method === 'PATCH') return 'update';
  if (method === 'DELETE') return 'delete';
  if (method === 'GET' && path.includes('/export')) return 'export';
  return 'view';
}

function extractModule(path: string): string {
  if (path.includes('/users')) return 'user';
  if (path.includes('/stations')) return 'station';
  if (path.includes('/piles')) return 'pile';
  if (path.includes('/orders')) return 'order';
  if (path.includes('/alarms')) return 'alarm';
  return 'other';
}
```

```typescript
// apps/server/src/utils/exportUtil.ts
import ExcelJS from 'exceljs';
import { Response } from 'express';

export interface ExportColumn {
  header: string;
  key: string;
  width?: number;
}

export async function exportToExcel(
  res: Response,
  data: any[],
  columns: ExportColumn[],
  filename: string
) {
  const workbook = new ExcelJS.Workbook();
  const worksheet = workbook.addWorksheet('Sheet1');

  // 设置列
  worksheet.columns = columns.map(col => ({
    header: col.header,
    key: col.key,
    width: col.width || 15,
  }));

  // 设置表头样式
  worksheet.getRow(1).font = { bold: true };
  worksheet.getRow(1).fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: 'FFE0E0E0' },
  };

  // 添加数据
  data.forEach(item => {
    worksheet.addRow(item);
  });

  // 设置响应头
  res.setHeader(
    'Content-Type',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
  );
  res.setHeader(
    'Content-Disposition',
    `attachment; filename="${encodeURIComponent(filename)}.xlsx"`
  );

  // 写入响应
  await workbook.xlsx.write(res);
  res.end();
}
```

```typescript
// apps/admin/src/utils/export.ts
import axios from 'axios';

export async function exportData(
  url: string,
  params: any,
  filename: string
) {
  const response = await axios.post(
    url,
    params,
    {
      responseType: 'blob',
      headers: {
        Authorization: `Bearer ${getToken()}`,
      },
    }
  );

  const blob = new Blob([response.data], {
    type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
  });
  const downloadUrl = window.URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = downloadUrl;
  link.download = `${filename}.xlsx`;
  link.click();
  window.URL.revokeObjectURL(downloadUrl);
}
```

```typescript
// apps/server/src/services/recommendationService.ts

interface RecommendationScore {
  stationId: number;
  score: number;
  reasons: string[];
}

export async function getRecommendations(
  userId: number,
  latitude: number,
  longitude: number,
  limit: number = 10
): Promise<RecommendationScore[]> {
  // 1. 获取用户历史充电记录
  const userOrders = await Order.findAll({
    where: { user_id: userId },
    attributes: ['station_id'],
    group: ['station_id'],
  });
  
  const visitedStations = userOrders.map(o => o.station_id);
  
  // 2. 获取附近充电站
  const nearbyStations = await Station.findAll({
    attributes: ['id', 'name', 'latitude', 'longitude'],
    // 这里需要根据经纬度计算距离，简化示例
  });
  
  // 3. 计算推荐分数
  const recommendations: RecommendationScore[] = [];
  
  for (const station of nearbyStations) {
    let score = 0;
    const reasons: string[] = [];
    
    // 距离分数（越近越高）
    const distance = calculateDistance(latitude, longitude, station.latitude, station.longitude);
    const distanceScore = Math.max(0, 100 - distance * 10);
    score += distanceScore * 0.3;
    if (distance < 5) reasons.push('距离很近');
    
    // 历史偏好分数（用户常去的站点加分）
    if (visitedStations.includes(station.id)) {
      score += 50 * 0.3;
      reasons.push('您常去此站');
    }
    
    // 评分分数
    const avgRating = await getStationAverageRating(station.id);
    score += avgRating * 10 * 0.2;
    if (avgRating >= 4.5) reasons.push('评分很高');
    
    // 空闲率分数
    const availabilityRate = await getStationAvailabilityRate(station.id);
    score += availabilityRate * 0.2;
    if (availabilityRate > 0.5) reasons.push('空闲率高');
    
    recommendations.push({
      stationId: station.id,
      score,
      reasons,
    });
  }
  
  // 4. 排序并返回Top N
  return recommendations
    .sort((a, b) => b.score - a.score)
    .slice(0, limit);
}

function calculateDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
  // 简化版距离计算（实际应该使用Haversine公式）
  const R = 6371; // 地球半径（公里）
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}
```

```typescript
export interface UsagePrediction {
  pileId: number;
  date: string;
  hour: number;
  predictedUsage: number;  // 0-1，使用率预测
  confidence: number;      // 0-1，置信度
}

export async function predictPileUsage(
  pileId: number,
  date: string
): Promise<UsagePrediction[]> {
  // 1. 获取历史使用数据（最近30天）
  const historicalData = await getHistoricalUsage(pileId, 30);
  
  // 2. 按小时聚合数据
  const hourlyData = aggregateByHour(historicalData);
  
  // 3. 预测模型（简化版：使用平均值+趋势）
  const predictions: UsagePrediction[] = [];
  
  for (let hour = 0; hour < 24; hour++) {
    const hourData = hourlyData[hour] || [];
    
    // 计算平均使用率
    const avgUsage = hourData.reduce((sum, d) => sum + d.usageRate, 0) / hourData.length || 0;
    
    // 计算趋势（最近7天 vs 之前7天）
    const recentWeek = hourData.slice(-7);
    const previousWeek = hourData.slice(-14, -7);
    const recentAvg = recentWeek.reduce((sum, d) => sum + d.usageRate, 0) / recentWeek.length || 0;
    const previousAvg = previousWeek.reduce((sum, d) => sum + d.usageRate, 0) / previousWeek.length || 0;
    const trend = recentAvg - previousAvg;
    
    // 预测值 = 平均值 + 趋势修正
    const predictedUsage = Math.max(0, Math.min(1, avgUsage + trend * 0.3));
    
    // 置信度 = 数据量 / 30（数据越多，置信度越高）
    const confidence = Math.min(1, hourData.length / 30);
    
    predictions.push({
      pileId,
      date,
      hour,
      predictedUsage,
      confidence,
    });
  }
  
  return predictions;
}
```
